\sect{Agentic RL}

\begin{frame}{The sandbox is the reward source: four tiers close one on-policy loop}
\flabel{fr:arch}
\figcap{Agentic RL: trainer, inference, harness, sandbox}
\centering
\figcard[height=0.46\textheight]{lanes_fig02_rl-loop.pdf}
\par\vspace{2mm}
\raggedright
\begin{itemize}\footnotesize\setlength{\itemsep}{0.8mm}
\item \textbf{Trainer} (slime: Megatron, GRPO) updates the policy; \textbf{inference} (SGLang engines) serves it.
\item \textbf{Harness} runs the agent loop; a proxy records the exact \tc{token\_ids} + logprobs (\hl{TITO}), so training stays on-policy.
\item \textbf{Sandbox} (AgentCore) runs the tool calls and the checker: \hl{the sandbox is the reward source}.
\end{itemize}
\end{frame}

%====================================================================
\begin{frame}{Each episode becomes one loss-masked Sample, and rollout runs async with training}
\flabel{fr:pipeline}
\figcap{slime pipeline: rollout $\rightarrow$ data buffer $\rightarrow$ training $\rightarrow$ weight sync}
\centering
\figcard[width=0.72\textwidth]{fig15_slime-architecture.png}
\par\vspace{3mm}
\raggedright
\begin{enumerate}\footnotesize\setlength{\itemsep}{0.6mm}
\item \textbf{Seed:} the harness opens a sandbox session and writes the task's files.
\item \textbf{Act $\rightarrow$ observe:} policy turn $\rightarrow$ tool call runs in the sandbox $\rightarrow$ output fed back, for several turns.
\item \textbf{Check:} a hidden checker runs in the sandbox $\rightarrow$ binary reward (RLVR, no LLM judge).
\item \textbf{Pack:} one loss-masked \tc{Sample} per episode $\rightarrow$ data buffer.
\item \textbf{Train + sync:} a GRPO step, weights pushed to SGLang; rollout runs \hl{async} with training.
\end{enumerate}
\end{frame}

\sect{AgentCore foundations}

\begin{frame}{The same loop code runs on either substrate; only the Runtime ARN differs}
\flabel{fr:substrates}
\begin{itemize}\footnotesize\setlength{\itemsep}{1mm}
\item \textbf{Runtime:} your image, serverless, \hl{per-session isolated compute}, called with \tc{InvokeAgentRuntime} (SigV4).
\item \textbf{Code Interpreter:} a managed CPU sandbox, no image to build.
\item Same loop code on both substrates; only the Runtime ARN differs. \textbf{Both demos use microVMs.}
\end{itemize}
\vspace{2mm}

%====================================================================
\begin{frame}{A cold launch costs 0.77 s to minutes; a reused or pre-opened session opens in milliseconds}
\flabel{fr:lifecycle}
\begin{itemize}\footnotesize\setlength{\itemsep}{1mm}
\item \textbf{1 : N : 1.} Runtime = one image, one ARN; session = one \tc{runtimeSessionId} = one microVM / instance. \hl{Runtimes track images, not tasks.}
\item \textbf{Launch:} no ``create session'' API; the first invoke under a new id cold-starts the compute.
\item \textbf{End:} idle timeout, max lifetime, or \tc{StopRuntimeSession}.
\end{itemize}

\sect{Using AgentCore in slime}

\begin{frame}[fragile]{slime needs only its two existing hooks to use AgentCore, so there is no fork}
\flabel{fr:hook}
{\footnotesize\tc{--custom-generate-function-path} $\rightarrow$ \tc{generate.generate} (dispatches the episode, records turns); \tc{--custom-rm-path} $\rightarrow$ the reward.\par}
\vspace{1mm}
\begin{lstlisting}[language=Python]
async def generate(args, sample, sampling_params):        # slime hook
    task, pool = task_from(sample), harness_pool.get_pool(args)
    async def episode(worker):                            # a free dispatch worker pulls it
        async with pool.sandbox(key, open_session) as (sb, wait, slot):  # sandbox first
            with pool.engine() as (e, engine):            # then the least-loaded engine
                return await mini_claude_episode.run(..., sb, engine)  # loop runs IN sb
    return await pool.run(episode)                        # ONE loss-masked Sample
\end{lstlisting}
\begin{itemize}\footnotesize\setlength{\itemsep}{0.6mm}
\item Each sandbox op = one \tc{invoke\_agent\_runtime(ARN, session\_id)} in a thread $\rightarrow$ \hl{N episodes concurrently}.
\item \tc{\_\_aexit\_\_} ends the session when the episode returns: one teardown path.
\item \tc{SANDBOX=agentcore-runtime} $\rightarrow$ your own Runtime: the agent loop runs \textbf{in the session} and dials the trainer back, which the Code Interpreter can't.
\end{itemize}
\end{frame}

\begin{frame}{Deploying a Runtime takes three steps, and only the dial-back path needs ingress}
\flabel{fr:deploy}
\begin{columns}[T]
\column{0.52\textwidth}
\ptitle{Build, IAM, wire}
\begin{enumerate}\footnotesize\setlength{\itemsep}{1mm}
\item \textbf{Build + deploy:} image (\textbf{arm64} microVM / \textbf{amd64} Instance) $\rightarrow$ ECR $\rightarrow$ \tc{CreateAgentRuntime}. The image serves \tc{sandbox\_server\_agent.py}: \tc{write\_file} / \tc{exec} / \tc{read\_file}.

\begin{frame}{Pre-opening cut the p50 open wait to $\le$ 5 ms and the makespan by 15--20\,\%}
\flabel{fr:preopen}
\begin{enumerate}\footnotesize\setlength{\itemsep}{0.8mm}
\item \textbf{Look-ahead data source} (\tc{tb\_lookahead\_source.py}, a \tc{--data-source-path} plugin) replays slime's deterministic prompt order $\rightarrow$ the next 2 rollouts' prompts. \hl{No slime change.}
\item \textbf{Bounded window:} \tc{TB\_PREOPEN\_BATCHES=2} $\rightarrow$ \textbf{P = 64} sessions booted ahead, pooled per task; each claim starts the next boot. \tc{1} $\rightarrow$ 32, \tc{0} $\rightarrow$ off.
\item \textbf{Sizing:} boot hidden when P $\cdot$ E / W $\ge$ boot time; beyond that is idle cost. Tune by \tc{open\_wait\_ms} ($\gg$ 0 $\rightarrow$ raise) vs \tc{preopen\_idle\_ms} (the price).
\end{enumerate}

\sect{Demo: harness scaling}

\begin{frame}{The harness must scale with inference and the sandbox, or it silently becomes the ceiling}
\flabel{fr:scaleuse}
\begin{itemize}\footnotesize\setlength{\itemsep}{2mm}
\item \textbf{Use case covered:} scale an agentic-RL rollout by sizing \hl{inference, harness and sandbox independently}; the decoupled harness tier (\tc{harness\_pool.py}) is the \textbf{default in every demo}, and \tc{SCALE=1} turns all three up together (4 : 256 : 256, DeepScaleR on a stdlib image; \tc{demo\_cpu\_scale\_lanes} was retired into this preset, 2026-09-28).
\item \textbf{Why this demo:} (1) the harness must scale \textbf{along with} SGLang and the sandbox, or a few coroutines silently become the ceiling; (2) it should scale \textbf{with the sandbox, not as part of the trainer}: the agent loop runs in the episode's session, as in DeepSeek DSec.
\item \textbf{Design considerations:} \hl{throughput = min(inference, harness, sandbox)}; each tier is its own pool, joined per episode (\sref{fr:tiers}). Pinned together (worker = engine = microVM), the slowest pin sets the pace.
\end{itemize}
\src{\tc{code/demos/demo\_cpu\_tasks} (\tc{SCALE=1} preset), checked 2026-09-28.}
\end{frame}

\begin{frame}{The agent loop runs inside the sandbox session, so the harness tier scales with the sandbox}
\flabel{fr:inloop}
\begin{columns}[T]
\column{0.58\textwidth}
\begin{itemize}\footnotesize\setlength{\itemsep}{1.4mm}
\item \textbf{One agent loop per sandbox, in the same session}, in every demo, like DSec (only exception: \tc{demo\_cpu\_branching} runs the real Claude Code CLI in its microVM).
\item \textbf{Per episode:} write + start \tc{session\_runner.py} $\rightarrow$ long-poll $\rightarrow$ read transcript $\rightarrow$ \textbf{kill the runner, then the hidden grader} runs (the loop can't touch its reward).
\item \textbf{Model calls dial back} (VPC, SG$\rightarrow$SG) to the engine's \tc{AnthropicAdapter}, which records the exact tokens (TITO).
\item \hl{So the harness tier scales with the sandbox tier}: trainer-side dispatchers only start and poll.
\item \textbf{No fallback:} no \tc{python3} or no path back $\rightarrow$ \tc{HarnessPlacementError}; the loop never moves into the trainer.
\end{itemize}
\column{0.38\textwidth}
\centering

\begin{frame}{DSec's sandbox lifetimes match our idle timeout, so heartbeat and lifetime metrics are now default}
\flabel{fr:dsectake}
\begin{itemize}\footnotesize\setlength{\itemsep}{2mm}
\item \textbf{Their median sandbox lives about as long as our idle timeout} $\rightarrow$ heartbeat on by default; idle timeout 3600 s.
\item \textbf{New metrics, every demo:} \tc{perf/sandbox\_ready\_span\_s}, \tc{perf/sandbox\_open\_peak\_per\_s} (vs 25/s), per-session lifetime + longest idle gap (\tc{<log>.sessions.csv}).
\item \textbf{Terminal-Bench, re-parsed:} 414 sessions, lifetime p50 \hl{471 s} / p95 \hl{1841 s}; 55 outlived 900 s (kept by heartbeat); peak \textbf{36 opens/s}, \textbf{0 throttles}.
\item \textbf{Not exposed by AgentCore:} image layers, session pause, thousands of creates/s.
\item \textbf{Copied:} agent loop in the sandbox session (\sref{fr:inloop}). \textbf{Still to copy:} per-task network allowlist.
\end{itemize}
\src{Terminal-Bench smoke (2026-09-23) re-parsed with the new session metrics; DSec arXiv 2609.22978; report Ch.~7.}
\end{frame}

%====================================================================
\begin{frame}{Start generic, add images only when needed, and scale each tier on its own signal}
\flabel{fr:takeaways}
\begin{itemize}\small\setlength{\itemsep}{2.2mm}
\item \textbf{Start on one generic stdlib Runtime}: most literature tasks need no custom image (the Code Interpreter has the packages, but can't host the in-session agent loop).
\item \textbf{Add an image only when the environment differs}; route by \tc{task.image}, AgentCore sees only an image and a session.
\item \textbf{Hide boots with pre-open, bridge stalls with heartbeat} on your own Runtimes (\srefs{fr:preopen}{fr:heartbeat}).
\item \textbf{Scale the tiers independently}: the harness tier with the sandbox, not in the trainer; diagnose the binding tier (\tc{qdepth} / \tc{sandbox\_wait\_ms} / \tc{engine\_inflight}) before adding capacity.
\item \hl{Isolation = the session}: fresh microVM when tasks mutate the OS; pooled microVM + fresh workdir for workdir-only tasks.
\end{itemize}
{\scriptsize\color{footc}Sources for every number: \sref{fr:refs}.}
\end{frame}
