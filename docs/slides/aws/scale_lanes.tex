\sect{3 $\cdot$ The solution: decoupled tiers}

\begin{frame}{4 engines : 256 workers : 256 microVMs, joined per episode, nothing pinned}
\flabel{fr:decoupled}
\begin{columns}[T]
\column{0.6\textwidth}
\figcap{Each tier is its own pool; nothing is pinned}
\figcard[width=0.95\linewidth]{lanes_fig12_decoupled.pdf}
\column{0.37\textwidth}
\vspace{8mm}
\begin{itemize}\small\setlength{\itemsep}{2mm}
\item Free worker $\rightarrow$ any free microVM (fresh \cc{ep-<hex>/}) $\rightarrow$ least-loaded engine.
\item \cc{RB $\times$ NS} = \hl{256 episodes/step}.
\end{itemize}
\end{columns}
\end{frame}

\begin{frame}{Each tier has one knob, and the log names the tier that binds}
\flabel{fr:knobs}
\tabcap{One knob per tier; the log names the tier that binds}
\footnotesize
\begin{tabularx}{\textwidth}{@{}p{0.3\textwidth}p{0.14\textwidth}X@{}}
\toprule
\textbf{Knob} & \textbf{Tier} & \textbf{It binds when}\\
\midrule
\cc{HARNESS\_WORKERS} (256) & harness & \cc{qdepth} $>$ 0: episodes wait for a worker\\
\cc{SANDBOX\_SESSIONS} (256) & sandbox & \cc{sandbox\_wait\_ms} $>$ 0: workers wait for a microVM\\
\cc{ROLLOUT\_GPUS} (4) & inference & \cc{engine\_inflight} and \cc{sglang\_gen\_ms} climb\\
\bottomrule
\end{tabularx}
\par\vspace{4mm}
\begin{itemize}\small
\item Harness and sandbox scale for \textbf{different reasons} (Python concurrency vs microVM quota and cost), so they are two numbers, not one.
\end{itemize}
\vspace{2mm}
\plainen{if episodes queue, add workers; if workers queue, add microVMs; if generation slows, add GPUs.}
\src{Knob defaults of the decoupled pool (\cc{harness\_pool.py}), as of 2026-09-25.}
\end{frame}

\begin{frame}{A reused microVM serves one episode at a time; safe for DeepScaleR, not OS-level tasks}
\flabel{fr:isolation}
\begin{itemize}\small\setlength{\itemsep}{3mm}
\item One session serves \hl{one episode at a time}; a free-list hands it out, and it comes back only after \cc{rm -rf} of the episode's workdir.
\item Different sessions are different microVMs (AgentCore's isolation boundary).
\item Not reset between episodes of one session: \cc{/tmp}, packages, stray processes.
\begin{itemize}
\item Safe for DeepScaleR (writes only its workdir).
\item OS-level tasks need \cc{SANDBOX\_SESSIONS=0} (a fresh session per episode).
\end{itemize}
\end{itemize}
\vspace{4mm}
\plainen{each episode gets a clean folder on a machine nobody else is using right now; anything outside that folder carries over, so tasks that change the OS get a fresh machine instead.}
\end{frame}

\begin{frame}{256 microVMs open in about 18 s, well inside the quotas, once the client is widened}
\flabel{fr:opening}
\begin{itemize}\small\setlength{\itemsep}{3mm}
\item All sessions open once at start, paced at \hl{20/s} (quota 25 new/s [Q2]): ready in $\approx$ 13 s + one boot ($\approx$ 5 s).
\item Data-plane invokes: $\approx$ 100/s at 256 episodes vs a 1000/s quota. Active sessions: 256 vs 5000.
\item Client side: boto3 \cc{max\_pool\_connections} = 272 (default 10) and a 288-thread executor (default $\le$ 32), or the harness process becomes the ceiling.
\end{itemize}
\vspace{3mm}
\tabcap{Load vs quota at 256 episodes}
\footnotesize
\begin{tabular}{@{}lrr@{}}
\toprule
\textbf{Limit} & \textbf{This demo} & \textbf{Quota / default}\\
\midrule

\flabel{fr:pipeline}
\figcap{slime's async driver overlaps train N with rollout N+1}
\centering
\figcard[width=0.66\textwidth]{lanes_fig06_pipeline.pdf}\par
\vspace{3mm}
\raggedright
\begin{itemize}\small
\item Weight sync waits for the rollout to finish, so rollouts are \hl{one step off-policy}.
\end{itemize}

\begin{frame}{DeepSeek DSec moved the agent loop out of the trainer to survive GPU preemption [31]}
\flabel{fr:dsec}
\begin{itemize}\footnotesize\setlength{\itemsep}{1.5mm}
\item \textbf{Why:} DeepSeek's GPU jobs are routinely \textbf{preempted}. With the loop in the GPU pod, the loop died but the sandbox survived, so recovery \textbf{replayed a command log}.
\item \textbf{Where:} from V4.1 on, the harness runs on DSec, \textbf{outside the GPU pool}: an agent sandbox (scaffold + tools) and a worker container (rollout state). A preempted trainer \textbf{reconnects}, with no replay.
\end{itemize}
\vspace{2mm}
\tabcap{Agent-loop placement, DSec vs this demo}
\footnotesize
\begin{tabularx}{\textwidth}{@{}p{0.2\textwidth}XX@{}}
\toprule
 & \textbf{DSec (V4.1+)} & \textbf{This demo}\\
\midrule
Harness & agent sandbox, with its tools & \hl{slime's rollout process}\\
Token record & worker container & slime's in-process adapter\\
Trainer lost & reconnect, continue & abort in-flight: one batch\\
Trainer stalls & pause/resume sandbox & \hl{heartbeat} keeps sessions\\
\bottomrule
\end{tabularx}
\src{[31] DSec (DeepSeek Elastic Compute), arXiv 2609.22978, \S6.2--6.3; see \sref{fr:refs}.}
\end{frame}

\begin{frame}{Without preemption, moving the harness out of slime saves nothing, so it stays in-process}
\flabel{fr:inprocess}
\begin{itemize}\footnotesize\setlength{\itemsep}{1.2mm}
\item \textbf{No preemption here:} the trainer has a dedicated 8$\times$L40S box. A stall is the real risk, and the heartbeat covers it. A crash costs one batch, not correctness.
\item \textbf{Moving only the harness saves nothing:} the token record and the SGLang engines die with the slime job anyway. DSec's gain also needs an external record and a ``reconnect to episode X'' path \textbf{in slime}.
\item \textbf{Harness in each microVM has costs:}
\begin{itemize}
\item the policy's commands would run next to its own transcript and reward, so it could tamper with them;
\item every microVM needs a network path back to inference (VPC mode plus a security-group rule);
\item it re-ties the harness tier to the sandbox tier, which is the coupling this demo removes.
\end{itemize}
\item \textbf{When to move it:} preemptible/Spot GPUs or multi-hour episodes. Then run the harness plus its token record as its own service, and have slime fetch episodes by ID.
\end{itemize}
\end{frame}

\begin{frame}{Takeaways: decouple the tiers, then scale whichever one binds}
\flabel{fr:takeaways}
\vspace{4mm}
\begin{itemize}\normalsize\setlength{\itemsep}{5mm}
\item \textbf{AgentCore = the sandbox tier:} Runtime = environment, session = episode, microVM = isolation.
\item \textbf{Decouple the tiers:} harness workers, microVMs and engines are three pools joined per episode; persistent microVMs gave \hl{100$\times$ fewer opens} and \hl{3.6$\times$ lower sandbox p95}.
\item \textbf{Scale the bottleneck:} math is inference-bound; the 256-sandbox demo needs a sandbox-heavy task.
\end{itemize}
\src{\cc{\_smoke\_logs/lanes\_tb\_0923/} (2026-09-23): 4 vs 400 opens, p95 431 $\rightarrow$ 120 ms.}
\end{frame}
