\sect{1 \textperiodcentered{} Background: agentic RL, AgentCore, slime}
\begin{frame}{Agentic RL: many tool turns on a machine, scored by its end state}
\flabel{fr:agrl}
\begin{columns}[T]
\column{0.46\textwidth}
\figcap{The agentic RL loop}
\centering\figcard[width=0.9\linewidth]{tb_fig02_rl-loop.pdf}\par
\vspace{1.5mm}\raggedright
\begin{enumerate}\scriptsize\setlength{\itemsep}{0.5mm}
\item \textbf{Rollout:} the policy reads the task, calls a tool (\tb{bash}), reads the output, calls again, \dots
\item \textbf{Reward:} the environment checks the \hl{final machine state} (hidden tests).
\item \textbf{Update:} GRPO compares the N tries of one task; better-than-average tries are reinforced.
\end{enumerate}
\column{0.5\textwidth}
\tabcap{Single-turn RL vs agentic RL}
\tiny\setlength{\tabcolsep}{3pt}
\begin{tabularx}{\linewidth}{@{}P{0.2\linewidth}P{0.31\linewidth}Y@{}}
\toprule
& \textbf{Single-turn (math, code)} & \textbf{Agentic (Terminal-Bench)}\\
\midrule
Action & one answer & up to 8 turns of tool calls\\
Environment & none, or a checker & a stateful Linux machine\\
Reward & match the answer & hidden tests on the end state\\
Cost per rollout & GPU only, seconds & GPU + sandbox, \textbf{25\,s p50}\\
Isolation & not needed & \hl{one machine per try}\\
\bottomrule
\end{tabularx}
\par\vspace{2mm}
\tabcap{What the sandbox must give an RL batch}
\tiny
\begin{tabularx}{\linewidth}{@{}P{0.3\linewidth}Y@{}}
\toprule
Same start state & all N tries of a task start from one image\\
Isolation & one try cannot see another's changes\\
Scale & 32 machines per step, 96 alive while training (\sref{fr:conc})\\
Low latency & a boot on the critical path stalls the step\\
\bottomrule
\end{tabularx}
\par\vspace{1mm}
{\tiny\textbf{GRPO caveat:} if all N tries get the same reward, the group has zero advantage and teaches nothing (\sref{fr:signal}).\par}
\end{columns}
\src{\tb{run.sh} (RB = 4 tasks $\times$ NS = 8 tries, 8 turns); smoke run 2026-09-30; checked 2026-10-01.}
\end{frame}

\begin{frame}{AgentCore Runtime: one image per ARN, one microVM per session}
\flabel{fr:agentcore}
\begin{columns}[T]
\column{0.5\textwidth}
\figcap{Image $\rightarrow$ Runtime $\rightarrow$ sessions $\rightarrow$ microVMs}
\centering\figcard[width=0.88\linewidth]{tb_fig01_cardinality.pdf}\par
\vspace{1.5mm}\raggedright
{\footnotesize\textbf{What it is.}} {\scriptsize Amazon Bedrock AgentCore Runtime is a managed, serverless host for agent code. You give it an arm64 container image; it gives you an ARN. Each \tb{runtimeSessionId} you invoke gets a \hl{dedicated microVM}, torn down at the end.\par}
\vspace{1mm}
\tabcap{The three calls the demo uses}
\tiny\setlength{\tabcolsep}{3pt}
\begin{tabularx}{\linewidth}{@{}P{0.4\linewidth}Y@{}}
\toprule
\tb{CreateAgentRuntime} & once per task: image, VPC, role\\
\tb{InvokeAgentRuntime} & per op; a \textbf{new id's first call boots it}\\
\tb{StopRuntimeSession} & episode end: the microVM is gone\\
\bottomrule
\end{tabularx}
\column{0.46\textwidth}
\tabcap{AgentCore concepts seen from agentic RL}
\tiny\setlength{\tabcolsep}{3pt}
\begin{tabularx}{\linewidth}{@{}P{0.2\linewidth}P{0.34\linewidth}Y@{}}
\toprule
\textbf{AgentCore} & \textbf{RL meaning} & \textbf{One step here}\\
\midrule
\textbf{Image} & the environment (ECR, arm64, $\le$ 2\,GB) & one per task, 218 built\\
\textbf{Runtime} & a task's env behind one ARN & 4 tasks $\rightarrow$ 4 Runtimes\\
\textbf{Session} & one rollout & 4 $\times$ 8 = \hl{32 new sessions}\\
\textbf{microVM} & that rollout's isolation (2 vCPU, 8\,GB) & stopped when graded\\
GRPO group & 8 sessions on \textbf{one} Runtime & same start, nothing shared\\
\bottomrule
\end{tabularx}
\par\vspace{1mm}
\tabcap{Lifecycle facts that shape the RL design}
\tiny
\begin{tabularx}{\linewidth}{@{}P{0.4\linewidth}Y@{}}
\toprule
No ``create session'' call & first invoke = \textbf{cold start} $\rightarrow$ pre-open (\sref{fr:cold})\\
15-min idle reclaim, silent & id comes back empty $\rightarrow$ heartbeat (\sref{fr:hbwhy})\\
arm64 only, image $\le$ 2\,GB & 11 of 229 tasks can't ship (\sref{fr:build})\\
Quotas [Q1][Q2] & 5000 live sessions, 25 new/s, 1000 Runtimes\\
\bottomrule
\end{tabularx}
\end{columns}

\end{frame}

\column{0.46\textwidth}
\tabcap{One episode across the two sides}
\tiny\setlength{\tabcolsep}{3pt}
\begin{tabularx}{\linewidth}{@{}P{0.06\linewidth}P{0.2\linewidth}Y@{}}
\toprule
& \textbf{Where} & \textbf{What}\\
\midrule
1 & slime & \tb{generate} gets a row; a harness worker picks the least-loaded engine\\
2 & AgentCore & take the task's pre-opened session (\sref{fr:prehow})\\
3 & microVM & write \tb{session\_runner.py}; the mini\_claude loop runs \textbf{in the VM}\\
4 & VPC & each turn dials back to the engine's adapter (SG$\rightarrow$SG rule)\\
5 & microVM & inject hidden tests, run them: \hl{reward}\\

\bottomrule
\end{tabularx}
\par\vspace{1.5mm}
\begin{itemize}\tiny\setlength{\itemsep}{0.4mm}
\item \textbf{Why the loop runs in the VM:} the harness scales with the sandboxes, not the trainer.
\item \textbf{Why an adapter:} the loop speaks the Anthropic Messages API; slime still gets exact token ids.
\item \textbf{Eval} is slime's stock eval, run on its own loop by \tb{mode\_b\_eval}.
\end{itemize}
\end{columns}

Nothing beyond the base image & 14\\
\bottomrule
\end{tabularx}
\column{0.44\textwidth}
\textbf{\footnotesize Why not one shared image?}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{States contradict:} \tb{broken-python} deletes \tb{pip}; others need it.
\item \textbf{Size:} 48\,GB in total vs a 2\,GB cap.
\item \textbf{Reward validity:} tests read machine state.
\end{itemize}
\textbf{\footnotesize Why not install at episode start?}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item Builds take p50 6.8\,min and GRPO siblings could diverge.
\end{itemize}
\plainen{the task is not a question, it's a computer in a known state plus a list of checks.}

\begin{frame}[fragile]{Pre-open replays slime's data order and boots the next 64 episodes' sessions ahead}

\begin{lstlisting}[language=Python]
def get_samples(self, n):
    groups = super().get_samples(n)          # slime's own draw
    rb, ns = args.rollout_batch_size, args.n_samples_per_prompt
    future = self._peek(TB_PREOPEN_BATCHES * rb)   # no state moved
    drawn    = [arn_for(g[0])] * ns  for g in groups
    upcoming = [arn_for(s)]    * ns  for s in future
    pool.reserve(drawn, upcoming)            # boot ahead
    return groups

def _refill(self):              # on every draw and claim
    need = Counter((demand + future)[:P])    # P = 64
    for arn in need:            # open or stop per task
        while len(ready[arn]) < need[arn]:
            ready[arn].append(open_session(arn))
def _claim(self, arn):          # a worker starts an episode
    s = ready[arn].popleft(); self._refill(); return s
\end{lstlisting}
\column{0.43\textwidth}
\begin{enumerate}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{No slime change:} \tb{\_peek} replays slime's seeded shuffle.
\item \textbf{Window P} = 2 batches $\times$ RB $\times$ NS = \hl{64} booted sessions; wrong guesses are stopped.
\item \textbf{Each claim} starts one more boot.
\item \textbf{Failures retry} on a new id, never scored.
\end{enumerate}
\vspace{0.5mm}
{\tiny\textbf{Rejected:} reset a warm session (leaks) \textperiodcentered{} pool per task (1,328 idle VMs).\par}
\vspace{1mm}
\colorbox{panel}{\parbox{\dimexpr\linewidth-2\fboxsep}{\tiny Training holds W + P = 32 + 64 = 96 live microVMs; eval opens on top (peak 137, \sref{fr:conc}). Quota: \textbf{5000} active sessions, 25 new/s [Q2] (\sref{fr:refs}).}}
\end{columns}

\begin{frame}[fragile]{One process-wide keeper pings any session idle past half the timeout: 348 pings, 0 failed}

{\footnotesize\textbf{\tb{session\_keeper.py}} \textcolor{inkmuted}{(simplified)}}
\begin{lstlisting}[language=Python]
class SessionKeeper:                 # one per rollout process
    def register(self, sid, pinger, idle_timeout): ...
    def touch(self, sid):            # every real invoke
        self._sessions[sid].last = now()
    async def _run(self):            # background loop
        while True:
            for s in self._sessions.values():
                if now() - s.last >= s.idle_timeout * 0.5:
                    try:
                        await s.pinger()   # exec("true")
                        s.last = now()
                    except Exception:  # retry next tick,
                        s.ping_failures += 1   # clock NOT reset
            await asyncio.sleep(tick)
\end{lstlisting}
{\footnotesize\textbf{\tb{TerminalBenchPool.\_open\_session}}}
\begin{lstlisting}[language=Python]
async def _ping(s=sb): await s.exec("true", timeout=30)

\end{lstlisting}
