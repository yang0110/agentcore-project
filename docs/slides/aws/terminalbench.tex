\begin{frame}{One image per row works on AgentCore; pre-open and heartbeat hide its costs}

\scriptsize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.17\textwidth}Y@{}}
\toprule
\textbf{Goal} & RL-train Qwen3-4B (slime GRPO) on Terminal-Bench, every episode in its own AgentCore microVM\\
\textbf{Data} & 229 single-container tasks $\rightarrow$ \hl{218 images and Runtimes} $\rightarrow$ \hl{166 rows} (135 train, 31 eval), one row per task (\sref{fr:data})\\

\textbf{Heartbeat} & Pre-opened sessions idle up to 16.5\,min, past the 15-min reclaim: 17 of 382 would have been lost; with pings, \hl{0 lost} (\sref{fr:hbres})\\
\textbf{Smoke run} & Runs end to end with 0 infra errors; learning not yet: 4 of 382 solved, reward too sparse for GRPO $\rightarrow$ partial credit next (\sref{fr:smoke})\\

\bottomrule
\end{tabularx}

\begin{frame}{Terminal-Bench is the case a shared sandbox image can't cover}
\flabel{fr:usecase}
\tabcap{What this demo covers}
\footnotesize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.2\textwidth}Y@{}}
\toprule
\textbf{Use case covered} & Tasks that need \textbf{their own OS image} (packages, \tb{/etc}, services): hundreds of custom images, one per task, routed per episode\\
\textbf{Why this demo} & Most literature RLVR tasks run in one generic image; this is the case they can't: custom images \textbf{at scale}, where a cold start is paid per episode\\
\textbf{Isolation} & One episode = one \textbf{brand-new session} = one dedicated microVM, stopped (VM terminated, memory sanitized) when the episode ends. Nothing leaks between episodes or GRPO siblings\\
\textbf{Design considerations} & Strict image routing (unmapped task = error, no fallback) \textperiodcentered{} look-ahead pre-open hides the boot \textperiodcentered{} always-on heartbeat keeps pre-opened sessions past the 15-min idle timeout \textperiodcentered{} retry on a new id; infra failures never scored as wrong answers\\
\bottomrule
\end{tabularx}
\par\vspace{3mm}
\plainen{every exercise comes with its own computer; each attempt gets a brand-new copy of it, booted before the trainer asks and kept awake until it's used.}
\src{\tb{code/demos/demo\_cpu\_tasks/terminalbench/}; checked 2026-09-29.}
\end{frame}

\sect{Agentic RL on AgentCore}

\begin{frame}{The sandbox is the reward source; slime reaches AgentCore through two existing hooks}
\flabel{fr:arch}
\begin{columns}[T]
\column{0.52\textwidth}
\figcap{Agentic RL: trainer, inference, harness, sandbox}
\centering\figcard[width=0.9\linewidth]{lanes_fig02_rl-loop.pdf}\par
\vspace{1mm}\raggedright
\begin{itemize}\scriptsize\setlength{\itemsep}{0.6mm}
\item \textbf{Trainer} (slime: Megatron, GRPO) updates the policy; \textbf{inference} (SGLang) serves it.
\item \textbf{Harness} runs the agent loop and records the exact token ids + logprobs, so training stays on-policy.
\item \textbf{Sandbox} (AgentCore) runs the tool calls and the hidden tests: \hl{it produces the reward}.
\end{itemize}
\column{0.44\textwidth}
\tabcap{How slime plugs in (no fork)}
\scriptsize\setlength{\tabcolsep}{3pt}
\begin{tabularx}{\linewidth}{@{}P{0.42\linewidth}Y@{}}
\toprule
\textbf{slime flag} & \textbf{Our code}\\
\midrule
\tb{-{}-custom-generate-}\newline\tb{function-path} & \tb{generate.generate}: one episode in a fresh session $\rightarrow$ one loss-masked Sample\\
\tb{-{}-custom-rm-path} & reward = hidden tests' result from the sandbox\\
\tb{-{}-data-source-}\newline\tb{path} & \tb{LookaheadDataSource}: tells the pool what comes next (\sref{fr:prehow})\\
\bottomrule
\end{tabularx}
\par\vspace{2mm}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item Every sandbox op is one \tb{InvokeAgentRuntime(ARN, session id)}; the agent loop runs \textbf{in} the session and dials the trainer's adapter back over the VPC.
\item Rollout is \hl{fully async}: episodes keep running while slime trains and syncs weights.
\end{itemize}
\end{columns}
\src{\tb{code/agentcore\_slime\_demo/generate.py}, \tb{harness\_pool.py}; \tb{demo\_cpu\_tasks/run.sh}; checked 2026-09-29.}
\end{frame}

\item \textbf{Reward = 1 only if every hidden test passes} (\tb{run-tests.sh} exit 0). No LLM judge.
\item \textbf{The tests read machine state}, so the starting machine must be exactly the one the task author built.
\end{itemize}
\plainen{the task is not a question, it's a computer in a known state plus a list of checks.}

\textbf{\footnotesize Why not one shared image?}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.4mm}

\end{itemize}
\textbf{\footnotesize Why not install at episode start?}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.4mm}
\item Builds take minutes to hours (p50 6.8\,min) and hit the network; siblings could diverge.
\end{itemize}

\begin{frame}[fragile]{We leave the task's environment untouched and append one 54\,MB server layer}
\flabel{fr:wrap}
\begin{columns}[T]
\column{0.55\textwidth}
{\footnotesize\textbf{\tb{Dockerfile.acs}} \textcolor{inkmuted}{(task Dockerfile + this, generated per task)}}
\begin{lstlisting}[language=bash]
# ... the task's own Dockerfile, unchanged ...
FROM ghcr.io/laude-institute/t-bench/ubuntu-24-04:latest
RUN apt-get update && apt-get install -y acl util-linux
RUN groupadd -f research && useradd -m alice ...

# ===== sandbox wrapper (appended by deploy_terminalbench.py)
USER root
COPY --from=acs py /opt/acs/py            # standalone Python
COPY --from=acs acs_http_server.py sandbox_server_agent.py /opt/acs/
RUN /opt/acs/py/bin/python3 -c "import http.server, tarfile, json"
EXPOSE 8080
ENTRYPOINT []
CMD ["/opt/acs/py/bin/python3", "/opt/acs/acs_http_server.py"]
\end{lstlisting}
\column{0.41\textwidth}
\begin{itemize}\scriptsize\setlength{\itemsep}{1.2mm}
\item \textbf{Own Python, not the task's.} Task images range from Python 3.8 to no Python at all, and changing the task's Python would change what gets graded.
\item \textbf{A stdlib HTTP server} on port 8080 speaks AgentCore's invoke protocol: \tb{exec}, file put/get, boot timestamps.
\item \textbf{\tb{WORKDIR} kept} (\tb{/app}, \tb{/app/astropy}, \dots); the server reports it on the first ping.
\item \textbf{No tests in the image.} \tb{tests/} and \tb{run-tests.sh} are injected from the trainer after the agent stops.
\end{itemize}
\end{columns}

\begin{frame}[fragile]{Building the pool takes six commands; validate on two tasks first}
\flabel{fr:buildsteps}
\begin{columns}[T]
\column{0.58\textwidth}
{\footnotesize\textbf{From \tb{code/demos/demo\_cpu\_tasks/}, on the host}}
\begin{lstlisting}[language=bash]
# 0. once: arm64 emulation + a buildx builder (2_build_image.sh does it)
docker run --privileged --rm tonistiigi/binfmt --install arm64

# 1. fetch the benchmark -> terminalbench/tasks/ + manifest.json
python3 terminalbench/fetch_terminalbench.py

# 2. dry run on 2 tasks (real AWS $): build + push + register
VARIANT=terminalbench TB_LIMIT=2 bash 2_build_image.sh
VARIANT=terminalbench TB_LIMIT=2 bash 4_deploy_runtime.sh

# 3. every task: build + push to ECR acs-tb-<task> (16 parallel)
VARIANT=terminalbench bash 2_build_image.sh
VARIANT=terminalbench bash 3_push_ecr.sh     # prints images.csv

# 4. one Runtime per image -> tasks/arn_by_image.json
VARIANT=terminalbench bash 4_deploy_runtime.sh
\end{lstlisting}
\column{0.38\textwidth}
\begin{enumerate}\scriptsize\setlength{\itemsep}{1.2mm}
\item[0.] Needed only on an x86 host.
\item[1.] Clones the repo, keeps the 229 single-container tasks.
\item[2.] Catches wrapper and IAM problems in minutes, not hours.
\item[3.] Writes \tb{status} / \tb{size\_mb} / \tb{build\_s} per task to \tb{images.csv}; over-cap and failed builds are skipped, not fatal.
\item[4.] VPC Runtimes with NAT egress (the tests install \tb{uv} and pytest at grade time) and the SG$\rightarrow$SG rule to the adapter ports.

\small\textbf{We can't reuse a warm VM.} Terminal-Bench tasks change the whole OS (apt, \tb{/etc}, services, \tb{/tmp}). Resetting a workdir would leak state into the next episode and corrupt the reward and the GRPO advantage. \hl{The session is the isolation boundary}, so each episode needs a new one, and a new one can be cold.\par
\vspace{2mm}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.5mm}
\item \textbf{Bursty:} all 8 GRPO siblings of a task start at once on the same Runtime; about 1 finds a warm VM, \hl{7 boot cold} (\sref{fr:cold}).
\item \textbf{Costly:} a cold open adds 6.7\,s p50 to a 21\,s median episode (+32\,\%) and up to 49\,s for 1--2\,GB images.
\item \textbf{slime reveals little:} it only knows the episodes it has drawn; the sandbox pool can't see what comes next.
\end{itemize}
\vspace{1.5mm}
\tabcap{Options}
\scriptsize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.34\textwidth}ccY@{}}
\toprule
\textbf{Option} & \textbf{Hides boot?} & \textbf{Keeps isolation?} & \textbf{Catch}\\
\midrule
Reuse one warm session per task + reset & \yes & \no & OS state leaks $\rightarrow$ \textbf{rejected}\\
Standing warm pool per task & \yes & \yes & 166 tasks $\times$ NS idle VMs\\

\hl{Look-ahead pre-open (used)} & \yes & \yes & Guesses can be wrong; booted VMs idle\\

\end{frame}

\begin{frame}[fragile]{Pre-open replays slime's data order and boots the next 64 episodes' sessions}
\flabel{fr:prehow}
\begin{columns}[T]
\column{0.53\textwidth}
{\footnotesize\textbf{\tb{tb\_lookahead\_source.py}} \textcolor{inkmuted}{(a \tb{-{}-data-source-path} plugin)}}
\begin{lstlisting}[language=Python]
def get_samples(self, n):
    groups = super().get_samples(n)          # slime's own draw
    rb, ns = args.rollout_batch_size, args.n_samples_per_prompt
    future = self._peek(TB_PREOPEN_BATCHES * rb)   # no state moved
    drawn    = [arn_for(g[0])] * ns  for g in groups
    upcoming = [arn_for(s)]    * ns  for s in future
    pool.reserve(drawn, upcoming)            # boot ahead
    return groups
\end{lstlisting}
{\footnotesize\textbf{\tb{TerminalBenchPool}} \textcolor{inkmuted}{(simplified)}}
\begin{lstlisting}[language=Python]
def _refill(self):              # on every draw and claim
    need = Counter((demand + future)[:P])    # P = 64
    for arn in need:            # open or stop per task
        while len(ready[arn]) < need[arn]:
            ready[arn].append(open_session(arn))
def _claim(self, arn):          # a worker starts an episode
    s = ready[arn].popleft(); self._refill(); return s
\end{lstlisting}
\column{0.43\textwidth}
\begin{enumerate}\scriptsize\setlength{\itemsep}{1mm}
\item \textbf{Look ahead without a slime change.} \tb{\_peek} replays slime's deterministic order (offset + \tb{random.seed(seed + epoch)} shuffle) to list the next 2 batches.
\item \textbf{Bounded window.} Keep a fresh session booted for each of the next \textbf{P} = \tb{TB\_PREOPEN\_BATCHES} $\times$ RB $\times$ NS = \hl{64} episodes, pooled per task. Wrong guesses are stopped at the next draw.
\item \textbf{Claim moves the window.} Each claimed session starts booting one more.
\item \textbf{Retry on a new id} (3$\times$); infra failures are logged as \tb{infra\_error}, never scored.
\end{enumerate}
\vspace{1mm}
\colorbox{panel}{\parbox{\dimexpr\linewidth-2\fboxsep}{\scriptsize Live microVMs $\le$ W + P = 32 + 64. Quota: \textbf{5000} active sessions, 25 new/s [Q2] (\sref{fr:refs}).}}
\end{columns}
\src{\tb{agentcore\_slime\_demo/tb\_lookahead\_source.py}, \tb{terminalbench\_pool.py} (\tb{reserve}, \tb{\_refill}, \tb{\_claim}); checked 2026-09-29.}
\end{frame}

\sect{5 \textperiodcentered{} Heartbeat}

\begin{frame}{An idle session is silently reclaimed after 15 min, and pre-opened sessions idle that long}
\flabel{fr:hbwhy}
\begin{columns}[T]
\column{0.52\textwidth}
\textbf{\footnotesize What AgentCore does}
\begin{itemize}\scriptsize\setlength{\itemsep}{1mm}
\item A Runtime session with no invoke for \tb{idleRuntimeSessionTimeout} (default \textbf{15 min}) is reclaimed [D3] (\sref{fr:refs}).
\item \textbf{Silently:} the next invoke on the same session id succeeds, on a \hl{fresh, empty sandbox}. Files, installed packages and the agent's work are gone, and nothing errors.
\end{itemize}
\textbf{\footnotesize Why our sessions go idle}
\begin{itemize}\scriptsize\setlength{\itemsep}{1mm}
\item \textbf{Pre-open:} a session is booted ahead and waits for its episode.
\item \textbf{Weight sync:} while slime pushes new weights, generation stops and every open session waits.
\end{itemize}
\column{0.44\textwidth}

\scriptsize

\toprule

Idle past 450\,s (ping point) & 172\\
Idle past \textbf{900\,s} (reclaim) & \hl{17 (4.5\,\%)}\\
\bottomrule
\end{tabularx}

\plainen{without a keep-alive, 1 in 22 pre-opened machines would be wiped before its episode, and the episode would score 0 on an empty machine without anyone noticing.}

\begin{frame}[fragile]{One process-wide keeper pings any session idle past half the timeout}
\flabel{fr:hbhow}
\begin{columns}[T]
\column{0.53\textwidth}
{\footnotesize\textbf{\tb{session\_keeper.py}} \textcolor{inkmuted}{(simplified)}}
\begin{lstlisting}[language=Python]
class SessionKeeper:                 # one per rollout process
    def register(self, sid, pinger, idle_timeout): ...
    def touch(self, sid):            # every real invoke
        self._sessions[sid].last = now()
    async def _run(self):            # background loop
        while True:
            for s in self._sessions.values():
                if now() - s.last >= s.idle_timeout * 0.5:
                    try:
                        await s.pinger()   # exec("true")
                        s.last = now()
                    except Exception:  # retry next tick,
                        s.ping_failures += 1   # clock NOT reset
            await asyncio.sleep(tick)
\end{lstlisting}
{\footnotesize\textbf{\tb{TerminalBenchPool.\_open\_session}}}
\begin{lstlisting}[language=Python]
async def _ping(s=sb): await s.exec("true", timeout=30)
keeper.register(sb.session_id, _ping, WARM_IDLE_TIMEOUT)
\end{lstlisting}
\column{0.43\textwidth}
\begin{itemize}\scriptsize\setlength{\itemsep}{1.2mm}
\item \textbf{Outside the harness.} No agent loop grows its own keep-alive; every session is registered when it opens and unregistered when it stops.
\item \textbf{Cheap ping:} \tb{exec "true"}, one invoke that resets AgentCore's idle clock.
\item \textbf{Zero pings while busy:} real episode traffic refreshes the same clock, so only idle sessions are pinged.
\item \textbf{A failed ping doesn't reset the clock}, so it retries at the next tick instead of a full interval later.
\item \textbf{Always on} for Terminal-Bench rows; \tb{HEARTBEAT=1} (default) for the rest. New Runtimes also get \tb{RUNTIME\_IDLE\_TIMEOUT=3600} as a backstop.
\end{itemize}
\end{columns}
\src{\tb{agentcore\_slime\_demo/session\_keeper.py} (\tb{SessionKeeper.\_run}), \tb{terminalbench\_pool.py} (\tb{\_open\_session}); checked 2026-09-29.}
\end{frame}
