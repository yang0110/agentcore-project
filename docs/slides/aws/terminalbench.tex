\sect{2 \textperiodcentered{} The challenge: cold start}

\begin{frame}{On the critical path, fresh sessions would cost $\approx$\,8 min of serial boot per step}
\flabel{fr:boot}
\small\textbf{We can't reuse a warm VM.} Terminal-Bench tasks change the whole OS (apt, \tb{/etc}, services, \tb{/tmp}). Resetting a workdir would leak state into the next episode and corrupt the reward and the GRPO advantage. \hl{The session is the isolation boundary}, so each episode needs a new one.\par
\vspace{3mm}
\tabcap{Cold-start cost}
\footnotesize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.45\textwidth}Y@{}}
\toprule
\textbf{Cost of a fresh session} & \textbf{Typical}\\
\midrule
Cold boot: allocate VM, pull image, start server & \textbf{$\approx$\,15 s} for a 447 MB image; grows with image size\\
Warm hit: AgentCore already had a spare VM up & $\approx$\,0.5 s\\
Per step, if on the critical path & 32 $\times$ 15 s $\approx$ \hl{8 min} of serial boot\\
\bottomrule
\end{tabularx}
\par\vspace{3mm}
\begin{itemize}\footnotesize\setlength{\itemsep}{1mm}
\item \textbf{Bursty:} all 8 GRPO siblings of a task start at once, on the same Runtime.
\item \textbf{Idle reclaim:} a session silent for $\approx$\,15 min is reclaimed, and the weight-sync pause can be that long [D3] (\sref{fr:refs}).
\end{itemize}
\src{\tb{code/demos/demo\_cpu\_tasks/terminalbench/} session-open timings; idle timeout per [Q1], [D3]; checked 2026-09-25.}
\end{frame}

%====================================================================
\begin{frame}{Only look-ahead pre-open hides the boot, keeps isolation and stays on-policy}
\flabel{fr:options}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.5mm}
\item \textbf{slime only reveals a few episodes at a time:} at most \tb{CONC $\times$ engines} = 8 \tb{generate()} calls in flight.
\item \textbf{It draws step N+1 only after step N finishes}, so every step's first episodes would wait for a boot.
\item \textbf{slime knows nothing about sandboxes.} Tasks, Runtimes and sessions all live in our \tb{TerminalBenchPool}, inside slime's rollout process.
\end{itemize}
\vspace{1.5mm}
\tabcap{Options}
\scriptsize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.34\textwidth}ccY@{}}
\toprule
\textbf{Option} & \textbf{Hides boot?} & \textbf{Keeps isolation?} & \textbf{Catch}\\
\midrule
Reuse one warm session per task + reset & \yes & \no & OS state leaks $\rightarrow$ \textbf{rejected}\\
Standing warm pool per task & \yes & \yes & 218 tasks $\times$ NS idle VMs\\
Open on pickup & \no & \yes & Boot on the critical path\\
Fully-async rollout & \yes & \yes & Off-policy (stale weights); slime mode change\\
\hl{Look-ahead pre-open (used)} & \yes & \yes & Guesses can be wrong; booted VMs idle\\
\bottomrule
\end{tabularx}
\src{\tb{code/demos/demo\_cpu\_tasks/terminalbench/} rollout config (CONC, engines); checked 2026-09-25.}
\end{frame}

\sect{3 \textperiodcentered{} The solution: look-ahead pre-open}

\begin{frame}{Four mechanisms hide the boot with no slime change and at most 68 live microVMs}
\flabel{fr:mech}
\begin{enumerate}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{Look-ahead data source.} \tb{tb\_lookahead\_source.py} (a \tb{-{}-data-source-path} plugin) replays slime's deterministic order (sequential offset, \tb{random.seed(seed + epoch)} shuffle) to list the \textbf{next 2 rollouts'} prompts, without moving slime's position. No slime change.
\item \textbf{Bounded pre-open window.} The pool keeps a fresh session booted for each of the next \textbf{P} = \tb{TB\_PREOPEN\_BATCHES} $\times$ RB $\times$ NS = \hl{64} upcoming episodes, pooled per task. A worker claims any ready session for its task; the claim moves the window on by one. Wrong guesses are stopped at the next draw.
\item \textbf{Heartbeat.} A \tb{SessionKeeper} pings every open session at half the 15-min idle timeout, so none is reclaimed during training or weight sync.
\item \textbf{Retry on a new id.} A failed open retries on a \textbf{new} session (3$\times$); an infra failure is logged as \tb{infra\_error}, never scored as a wrong answer.
\end{enumerate}
\vspace{2mm}
\colorbox{panel}{\parbox{\dimexpr\linewidth-2\fboxsep}{\footnotesize Live microVMs $\le$ W + P = 4 + 64 = \hl{68}. The account quota is \textbf{5000} active sessions and 25 new/s [Q2] (\sref{fr:refs}); throttled opens back off and retry.}}
\src{\tb{code/demos/demo\_cpu\_tasks/terminalbench/} (\tb{tb\_lookahead\_source.py}, \tb{TerminalBenchPool}); quotas [Q2] us-west-2, 2026-09-23; checked 2026-09-25.}
\end{frame}

%====================================================================
\begin{frame}{Across a step boundary no worker waits for a boot; the price is idle VMs}
\flabel{fr:boundary}
\tabcap{Look-ahead across a step boundary (W = 4, P = 64)}
\footnotesize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.2\textwidth}Y P{0.17\textwidth}@{}}
\toprule
\textbf{Moment} & \textbf{What happens} & \textbf{Worker waits for a boot?}\\
\midrule
Step N$-$1 running & Its draw already announced steps N and N+1 $\rightarrow$ 64 sessions booting & ---\\
Step N drawn & Window = steps N, N+1: already booted, nothing new to open & ---\\
Each claim & Next episode of step N+2 starts booting & \hl{No}\\
Step N training & 64 booted VMs idle through training; heartbeat keeps them & ---\\
Step N+1 drawn & All its episodes already booted & \hl{No}\\
\bottomrule
\end{tabularx}
\par\vspace{4mm}
\begin{itemize}\footnotesize\setlength{\itemsep}{1.5mm}
\item \textbf{Sessions per Runtime are never configured.} A task sampled in a step gets NS = 8 sessions; others get 0.
\item \textbf{The price is idle time:} each VM is booted about P\,$\cdot$\,E\,/\,W seconds before it's needed (E = episode time).
\end{itemize}
\src{\tb{code/demos/demo\_cpu\_tasks/terminalbench/} pool design (W = 4, TB\_PREOPEN\_BATCHES = 2); checked 2026-09-25.}
\end{frame}

%====================================================================
\begin{frame}{One batch of look-ahead already hides the boot; the run's two cost $\approx$\,6 min idle per VM}
\flabel{fr:sizing}
\small Boot is hidden when \hl{P\,$\cdot$\,E\,/\,W $\ge$ B} (B = boot time), and a window of at least one batch (P $\ge$ RB $\times$ NS) also covers each step's first episodes. Everything beyond that is paid idle time.\par
\vspace{2mm}
\tabcap{Choosing TB\_PREOPEN\_BATCHES}
\footnotesize\setlength{\tabcolsep}{4pt}
\begin{tabularx}{\textwidth}{@{}P{0.27\textwidth}r Y P{0.25\textwidth}@{}}
\toprule
\textbf{\tb{TB\_PREOPEN\_BATCHES}} & \textbf{P} & \textbf{Boot hidden?} & \textbf{Idle per VM $\approx$ P\,$\cdot$\,E\,/\,W}\\
\midrule
0 --- look-ahead off & $\approx$\,4 & Mid-step only; each step's first W wait & $\approx$\,20 s\\
1 --- one batch ahead & 32 & Yes, including step starts & $\approx$\,3 min\\
\hl{2 --- used in the smoke run} & \hl{64} & Yes, with a full batch of slack & $\approx$\,6 min (measured: 7.3 min p50)\\
\bottomrule
\end{tabularx}
\par\vspace{3mm}
\begin{itemize}\footnotesize\setlength{\itemsep}{1mm}
\item E = 21 s (measured episode p50), W = 4.
\item Tune from the \tb{[demo]} lines: \tb{open\_wait\_ms} $\gg$ 0 $\rightarrow$ raise it; \tb{preopen\_idle\_ms} $\gg$ boot time $\rightarrow$ lower it.
\end{itemize}
\src{Smoke run 2026-09-23 (E, measured idle), \tb{code/demos/\_smoke\_logs/lanes\_tb\_0923}; checked 2026-09-25.}
\end{frame}

\begin{frame}{Heartbeat kept 3/3 own-Runtime sessions past the idle timeout; without it, 0/3 survived}
\flabel{fr:hb}
\small A Runtime session idle past \tb{idleRuntimeSessionTimeout} is reclaimed. In training that timeout is the \textbf{15-min default} (\tb{WARM\_IDLE\_TIMEOUT} = 900, pings at 450 s). The probe Runtime \tb{acs\_warmstart\_cpu} sets it to \textbf{60 s} only so the test fits in minutes; sessions then sit idle 150 s.\par

\begin{frame}{Cold start is solved but over-provisioned; learning signal is now the bottleneck}
\flabel{fr:next}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{Works end to end:} per-task Runtimes, one fresh microVM per episode, oracle-checked tasks, 0 infra failures.
\item \textbf{Cold start is solved, but over-provisioned.} Look-ahead removed all boot wait (the warm-start probe confirms: only the first batch waits), but at P = 64 a VM idles $\approx$\,7 min for a 21 s episode. With warm hits at $\approx$\,0.7 s, \hl{\tb{TB\_PREOPEN\_BATCHES} = 1 (or 0)} should give the same wait at a fraction of the cost.
\item \textbf{Learning signal is the bottleneck:} 4 solves in 382 episodes. Next: easy-first curriculum, more turns, or partial credit from \tb{tests\_passed}.
\end{itemize}
