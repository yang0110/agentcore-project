\column{0.44\textwidth}
\textbf{\footnotesize Why not one shared image?}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{States contradict:} \tb{broken-python} deletes \tb{pip}; others need it.
\item \textbf{Size:} 48\,GB in total vs a 2\,GB cap.
\item \textbf{Reward validity:} tests read machine state.
\end{itemize}
\textbf{\footnotesize Why not install at episode start?}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item Builds take p50 6.8\,min and GRPO siblings could diverge.
\end{itemize}
\plainen{the task is not a question, it's a computer in a known state plus a list of checks.}

\begin{frame}[fragile]{Pre-open replays slime's data order and boots the next 64 episodes' sessions ahead}
\flabel{fr:prehow}
\begin{columns}[T]
\column{0.53\textwidth}
{\footnotesize\textbf{\tb{tb\_lookahead\_source.py}} \textcolor{inkmuted}{(simplified; a \tb{-{}-data-source-path} plugin)}}
\begin{lstlisting}[language=Python]
def get_samples(self, n):
    groups = super().get_samples(n)          # slime's own draw
    rb, ns = args.rollout_batch_size, args.n_samples_per_prompt
    future = self._peek(TB_PREOPEN_BATCHES * rb)   # no state moved
    drawn    = [arn_for(g[0])] * ns  for g in groups
    upcoming = [arn_for(s)]    * ns  for s in future
    pool.reserve(drawn, upcoming)            # boot ahead
    return groups
\end{lstlisting}
{\footnotesize\textbf{\tb{TerminalBenchPool}} \textcolor{inkmuted}{(simplified)}}
\begin{lstlisting}[language=Python]
def _refill(self):              # on every draw and claim
    need = Counter((demand + future)[:P])    # P = 64
    for arn in need:            # open or stop per task
        while len(ready[arn]) < need[arn]:
            ready[arn].append(open_session(arn))
def _claim(self, arn):          # a worker starts an episode
    s = ready[arn].popleft(); self._refill(); return s
\end{lstlisting}
\column{0.43\textwidth}
\begin{enumerate}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{No slime change:} \tb{\_peek} replays slime's seeded shuffle.
\item \textbf{Window P} = 2 batches $\times$ RB $\times$ NS = \hl{64} booted sessions; wrong guesses are stopped.
\item \textbf{Each claim} starts one more boot.
\item \textbf{Failures retry} on a new id, never scored.
\end{enumerate}
\vspace{0.5mm}
{\tiny\textbf{Rejected:} reset a warm session (leaks) \textperiodcentered{} pool per task (1,328 idle VMs).\par}

\begin{lstlisting}[language=Python]
class SessionKeeper:                 # one per rollout process
    def register(self, sid, pinger, idle_timeout): ...
    def touch(self, sid):            # every real invoke
        self._sessions[sid].last = now()
    async def _run(self):            # background loop
        while True:
            for s in self._sessions.values():
                if now() - s.last >= s.idle_timeout * 0.5:
                    try:
                        await s.pinger()   # exec("true")
                        s.last = now()
                    except Exception:  # retry next tick,
                        s.ping_failures += 1   # clock NOT reset
            await asyncio.sleep(tick)
\end{lstlisting}
{\footnotesize\textbf{\tb{TerminalBenchPool.\_open\_session}}}
\begin{lstlisting}[language=Python]
async def _ping(s=sb): await s.exec("true", timeout=30)
keeper.register(sb.session_id, _ping, WARM_IDLE_TIMEOUT)
\end{lstlisting}
\column{0.43\textwidth}
\begin{itemize}\scriptsize\setlength{\itemsep}{0.8mm}
\item \textbf{Outside the harness:} registered on open, removed on stop.
\item \textbf{Cheap:} one \tb{exec "true"}; busy sessions are never pinged.
\item \textbf{A failed ping} retries next tick.
\end{itemize}
